	var top_h = clamp(74.0 * ui_scale, 64.0, 94.0)
	var bottom_h = clamp(166.0 * ui_scale, 142.0, 205.0)
	hud_rect = Rect2(
		Vector2(safe_rect.position.x, safe_rect.position.y),
		Vector2(safe_rect.size.x, top_h)
	)

	var maze_zone = Rect2(
		Vector2(safe_rect.position.x + 8.0, safe_rect.position.y + top_h + 10.0),
		Vector2(safe_rect.size.x - 16.0, max(120.0, safe_rect.size.y - top_h - bottom_h - 18.0))
	)
	if cols > 0 and rows > 0:
		tile_size = floor(min(maze_zone.size.x / float(cols), maze_zone.size.y / float(rows)))
		tile_size = max(18.0, tile_size)
		var maze_size = Vector2(cols * tile_size, rows * tile_size)
		maze_rect = Rect2(maze_zone.position + (maze_zone.size - maze_size) * 0.5, maze_size)

	joy_radius = clamp(size.y * 0.118, 78.0, 108.0)
	power_radius = clamp(size.y * 0.092, 66.0, 88.0)
	var bottom_y = safe_rect.end.y
	joy_center = Vector2(
		safe_rect.position.x + joy_radius + 18.0,
		bottom_y - joy_radius - 10.0
	)
	power_center = Vector2(
		safe_rect.end.x - power_radius - 22.0,
		bottom_y - power_radius - 20.0
	)
	var pause_size = clamp(58.0 * ui_scale, 54.0, 70.0)
	pause_rect = Rect2(
		Vector2(safe_rect.end.x - pause_size, safe_rect.position.y + (top_h - pause_size) * 0.5),
		Vector2.ONE * pause_size
	)
	queue_redraw()

func _safe_area_in_viewport(view_size: Vector2) -> Rect2:
	# Godot reports Android safe area in display pixels; touch/game input uses viewport coordinates.
	var screen_size_i = DisplayServer.screen_get_size(DisplayServer.SCREEN_OF_MAIN_WINDOW)
	var display_safe = DisplayServer.get_display_safe_area()
	if screen_size_i.x <= 0 or screen_size_i.y <= 0 or display_safe.size.x <= 0 or display_safe.size.y <= 0:
		return Rect2(Vector2.ZERO, view_size)
	var screen_size = Vector2(screen_size_i)
	var scale = Vector2(view_size.x / screen_size.x, view_size.y / screen_size.y)
	return Rect2(
		Vector2(display_safe.position) * scale,
		Vector2(display_safe.size) * scale
	)

func _process(delta: float) -> void:
	if state == "play" and not paused:
		_update_game(delta)
	_update_visuals(delta)
	queue_redraw()

func _input(event: InputEvent) -> void:
	if event is InputEventKey and event.pressed and not event.echo:
		var key := event.keycode
		if key == KEY_W or key == KEY_UP:
			_set_player_dir(Vector2.UP)
		elif key == KEY_S or key == KEY_DOWN:
			_set_player_dir(Vector2.DOWN)
		elif key == KEY_A or key == KEY_LEFT:
			_set_player_dir(Vector2.LEFT)
		elif key == KEY_D or key == KEY_RIGHT:
			_set_player_dir(Vector2.RIGHT)
		elif key == KEY_SPACE:
			_use_power()
		elif key == KEY_P or key == KEY_ESCAPE:
			_handle_back_or_pause()
		return

	if event is InputEventScreenTouch:
		if event.pressed:
			_touch_down(event.index, event.position)
		else:
			_touch_up(event.index, event.position)
	elif event is InputEventScreenDrag:
		_touch_drag(event.index, event.position)

func _touch_down(index: int, pos: Vector2) -> void:
	if state == "menu":
		_hit_button(pos, menu_buttons)
		return
	if state == "end":
		_hit_button(pos, end_buttons)
		return
	if paused:
		_hit_button(pos, pause_buttons)
		return
	if state != "play":
		return

	if pause_rect.has_point(pos):
		paused = true
		_haptic(12)
		return
	if pos.distance_to(joy_center) <= joy_radius * 1.45:
		joy_pointer = index
		joy_active = true
		_update_joystick(pos)
		_haptic(8)
	elif pos.distance_to(power_center) <= power_radius * 1.35:
		power_pressed = true
		_use_power()
		_haptic(16)

func _touch_drag(index: int, pos: Vector2) -> void:
	if joy_active and index == joy_pointer:
		_update_joystick(pos)

func _touch_up(index: int, _pos: Vector2) -> void:
	if index == joy_pointer:
		joy_pointer = -1
		joy_active = false
		joy_knob = Vector2.ZERO
		joy_dir = Vector2.ZERO
	power_pressed = false

func _update_joystick(pos: Vector2) -> void:
	var raw := pos - joy_center
	var max_len := joy_radius * 0.58
	var length := raw.length()
	if length < joy_radius * 0.13:
		joy_knob = Vector2.ZERO
		return

	var capped := raw.normalized() * min(length, max_len)
	joy_knob = capped

	var next_dir := Vector2.ZERO
	if abs(raw.x) > abs(raw.y) * 1.10:
		next_dir = Vector2.RIGHT if raw.x > 0.0 else Vector2.LEFT
	elif abs(raw.y) > abs(raw.x) * 1.10:
		next_dir = Vector2.DOWN if raw.y > 0.0 else Vector2.UP
	elif joy_dir == Vector2.ZERO:
		if abs(raw.x) >= abs(raw.y):
			next_dir = Vector2.RIGHT if raw.x > 0.0 else Vector2.LEFT
		else:
			next_dir = Vector2.DOWN if raw.y > 0.0 else Vector2.UP
	else:
		next_dir = joy_dir

	if next_dir != Vector2.ZERO and next_dir != joy_dir:
		joy_dir = next_dir
		_set_player_dir(next_dir)
		_haptic(5)

func _set_player_dir(d: Vector2) -> void:
	if reverse_timer > 0.0:
		d = -d
	if player.is_empty():
		return
	player.queued = d

func _handle_back_or_pause() -> void:
	if state == "play":
		paused = not paused
		_haptic(10)
	elif state == "end":
		state = "menu"
		_haptic(8)
	elif state == "menu":
		get_tree().quit()

func _start_game(new_mode: String) -> void:
	mode = new_mode
	state = "play"
	paused = false
	level = 1
	score = 0
	lives = 3
	time_left = 120.0
	inventory = ""
	shield = false
	speed_timer = 0.0
	magnet_timer = 0.0
	reverse_timer = 0.0
	ghost_rush_timer = 0.0
	double_score_timer = 0.0
	chaos_timer = rng.randf_range(7.0, 12.0)
	streak = 0
	streak_timer = 0.0
	power_mode = 0.0
	ghost_combo = 0
