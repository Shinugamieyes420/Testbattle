
func _draw_menu(size: Vector2) -> void:
	var center := Vector2(size.x * 0.5, size.y * 0.46)
	_text_center("GHOST GRID", center + Vector2(0, -190), 56, C_CYAN, min(800.0, size.x * 0.8))
	_text_center("Neon maze-chase, gebouwd voor Android touch.", center + Vector2(0, -145), 17, C_MUTED, min(800.0, size.x * 0.8))

	menu_buttons = []
	var bw := min(540.0, size.x * 0.46)
	var bh := 78.0
	var gap := 18.0
	var labels := [
		["CLASSIC RUN", MODE_CLASSIC, C_CYAN],
		["TIME ATTACK · 120 SEC", MODE_TIME, C_GOLD],
		["CHAOS MODE", MODE_CHAOS, C_PINK]
	]
	for i in range(labels.size()):
		var rect := Rect2(Vector2(center.x - bw * 0.5, center.y - 70 + i * (bh + gap)), Vector2(bw, bh))
		_draw_button(rect, labels[i][0], labels[i][2], false)
		menu_buttons.append({"rect": rect, "action": "start", "value": labels[i][1]})

	_text_center("Joystick links · Power rechts · Highscores worden automatisch opgeslagen", Vector2(center.x - size.x * 0.35, size.y - 62), 13, C_MUTED, size.x * 0.70)

func _draw_pause(size: Vector2) -> void:
	draw_rect(Rect2(Vector2.ZERO, size), Color(0,0,0,0.66), true)
	var c := size * 0.5
	_text_center("PAUZE", c + Vector2(0, -110), 52, C_TEXT, 500)
	pause_buttons = []
	var resume := Rect2(c + Vector2(-210, -38), Vector2(420, 72))
	var menu := Rect2(c + Vector2(-210, 54), Vector2(420, 72))
	_draw_button(resume, "VERDER", C_GREEN, false)
	_draw_button(menu, "TERUG NAAR MENU", C_RED, false)
	pause_buttons.append({"rect": resume, "action": "resume"})
	pause_buttons.append({"rect": menu, "action": "menu"})

func _draw_end(size: Vector2) -> void:
	draw_rect(Rect2(Vector2.ZERO, size), Color(0,0,0,0.72), true)
	var c := size * 0.5
	var title := "RUN COMPLETE" if mode == MODE_TIME and time_left <= 0.0 else "GAME OVER"
	_text_center(title, c + Vector2(0, -165), 54, C_PINK, 700)
	_text_center("SCORE " + str(score), c + Vector2(0, -95), 30, C_GOLD, 500)
	_text_center("LEVEL " + str(level) + "    ·    HI " + str(highscores.get(mode, 0)), c + Vector2(0, -55), 17, C_TEXT, 600)

	end_buttons = []
	var again := Rect2(c + Vector2(-230, 18), Vector2(460, 72))
	var menu := Rect2(c + Vector2(-230, 108), Vector2(460, 72))
	_draw_button(again, "NOG EEN KEER", C_GREEN, false)
	_draw_button(menu, "HOOFDMENU", C_CYAN, false)
	end_buttons.append({"rect": again, "action": "restart"})
	end_buttons.append({"rect": menu, "action": "menu"})

func _draw_center_overlay(text: String, color: Color) -> void:
	var size := get_viewport_rect().size
	draw_rect(Rect2(Vector2.ZERO, size), Color(0,0,0,0.38), true)
	_text_center(text, Vector2(size.x * 0.25, size.y * 0.53), 72, color, size.x * 0.50)

func _draw_button(rect: Rect2, label: String, color: Color, pressed: bool) -> void:
	var fill := Color(color, 0.31 if not pressed else 0.50)
	draw_rect(rect, fill, true)
	draw_rect(rect, Color(color, 0.80), false, 2.0)
	_text_center(label, rect.position + Vector2(0, rect.size.y * 0.62), 19, C_TEXT, rect.size.x)

func _hit_button(pos: Vector2, buttons: Array) -> void:
	for b in buttons:
		if b.rect.has_point(pos):
			_haptic(12)
			match b.action:
				"start":
					_start_game(b.value)
				"resume":
					paused = false
				"menu":
					state = "menu"
					paused = false
					_save()
				"restart":
					_start_game(mode)
			return

func _world_to_screen(pos: Vector2) -> Vector2:
	return maze_rect.position + pos * tile_size

func _text(text: String, pos: Vector2, size: int, color: Color) -> void:
	draw_string(font, pos, text, HORIZONTAL_ALIGNMENT_LEFT, -1, size, color)

func _text_right(text: String, pos: Vector2, size: int, color: Color) -> void:
	draw_string(font, pos, text, HORIZONTAL_ALIGNMENT_RIGHT, 0, size, color)

func _text_center(text: String, pos: Vector2, size: int, color: Color, width: float) -> void:
	draw_string(font, pos, text, HORIZONTAL_ALIGNMENT_CENTER, width, size, color)

func _best_highscore() -> int:
	return max(highscores[MODE_CLASSIC], highscores[MODE_TIME], highscores[MODE_CHAOS])

func _load_save() -> void:
	if not FileAccess.file_exists(SAVE_PATH):
		return
	var file := FileAccess.open(SAVE_PATH, FileAccess.READ)
	if file == null:
		return
	var parsed = JSON.parse_string(file.get_as_text())
	if typeof(parsed) == TYPE_DICTIONARY and parsed.has("highscores"):
		var h = parsed.highscores
		for k in highscores.keys():
			if h.has(k):
				highscores[k] = int(h[k])

func _save() -> void:
	var file := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	if file == null:
		return
	file.store_string(JSON.stringify({"highscores": highscores}))

func _exit_tree() -> void:
	_save()
