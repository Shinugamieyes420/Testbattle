func _draw_mobile_controls(size: Vector2) -> void:
	# Large thumb-first joystick. The interaction zone is larger than the visible circle.
	var joy_alpha = 0.42 if joy_active else 0.30
	draw_circle(joy_center, joy_radius * 1.13, Color(C_CYAN, 0.045))
	draw_circle(joy_center, joy_radius * 0.88, Color(0.02, 0.08, 0.16, 0.78))
	draw_arc(joy_center, joy_radius, 0, TAU, 64, Color(C_CYAN, joy_alpha), max(2.0, 3.0 * ui_scale), true)
	for d in DIRS:
		draw_line(joy_center + d * joy_radius * 0.48, joy_center + d * joy_radius * 0.72, Color(1,1,1,0.16), max(2.0, 2.0 * ui_scale))

	var knob = joy_center + joy_knob
	draw_circle(knob, joy_radius * 0.38, Color(C_CYAN, 0.14))
	draw_circle(knob, joy_radius * 0.30, C_CYAN)
	draw_circle(knob + Vector2(-joy_radius * 0.07, -joy_radius * 0.09), joy_radius * 0.07, Color(1,1,1,0.42))

	var ready = inventory != ""
	var power_color = C_PURPLE if ready else Color(0.22, 0.25, 0.36, 0.92)
	if ready:
		var pulse = 1.18 + 0.045 * sin(Time.get_ticks_msec() * 0.008)
		draw_circle(power_center, power_radius * pulse, Color(C_PURPLE, 0.11))
	draw_circle(power_center, power_radius, power_color)
	draw_arc(power_center, power_radius, 0, TAU, 56, Color(1,1,1,0.29), max(2.0, 3.0 * ui_scale), true)
	_text_center("POWER", power_center + Vector2(0, -8.0 * ui_scale), int(16 * ui_scale), C_TEXT, power_radius * 1.6)
	var power_name = _power_label(inventory) if ready else "LEEG"
	_text_center(power_name, power_center + Vector2(0, 17.0 * ui_scale), int(9 * ui_scale), Color(1,1,1,0.78), power_radius * 1.75)

	# Central status is kept above the Android gesture zone.
	var center_x = safe_rect.get_center().x if safe_rect.size.x > 0.0 else size.x * 0.5
	var status_y = min(joy_center.y + joy_radius * 0.52, safe_rect.end.y - 34.0 * ui_scale)
	if power_mode > 0.0:
		var bar_w = min(size.x * 0.25, 330.0 * ui_scale)
		var bar_pos = Vector2(center_x - bar_w * 0.5, status_y)
		draw_rect(Rect2(bar_pos, Vector2(bar_w, 11.0 * ui_scale)), Color(0,0,0,0.46), true)
		var ratio = clamp(power_mode / 8.8, 0.0, 1.0)
		draw_rect(Rect2(bar_pos, Vector2(bar_w * ratio, 11.0 * ui_scale)), C_PINK, true)
		_text_center("POWER MODE", Vector2(center_x, bar_pos.y - 8.0 * ui_scale), int(10 * ui_scale), C_PINK, bar_w)

	if streak >= 5:
		_text_center(str(streak) + " STREAK", Vector2(center_x, status_y - 32.0 * ui_scale), int(15 * ui_scale), C_GOLD, 230.0 * ui_scale)

func _draw_button(rect: Rect2, label: String, color: Color, pressed: bool) -> void:
	var fill := Color(color, 0.31 if not pressed else 0.50)
	draw_rect(rect, fill, true)
	draw_rect(rect, Color(color, 0.80), false, 2.0)
	_text_center(label, Vector2(rect.get_center().x, rect.position.y + rect.size.y * 0.62), int(18 * ui_scale), C_TEXT, rect.size.x * 0.92)

func _hit_button(pos: Vector2, buttons: Array) -> void:
	for b in buttons:
		if b.rect.has_point(pos):
			_haptic(12)
			match b.action:
				"start":
					_start_game(b.value)
				"resume":
					paused = false
				"menu":
					state = "menu"
					paused = false
					_save()
				"restart":
					_start_game(mode)
			return

func _world_to_screen(pos: Vector2) -> Vector2:
	return maze_rect.position + pos * tile_size

func _text(text: String, pos: Vector2, size: int, color: Color) -> void:
	draw_string(font, pos, text, HORIZONTAL_ALIGNMENT_LEFT, -1, size, color)

func _text_right(text: String, pos: Vector2, size: int, color: Color) -> void:
	var text_size = font.get_string_size(text, HORIZONTAL_ALIGNMENT_LEFT, -1, size)
	draw_string(font, Vector2(pos.x - text_size.x, pos.y), text, HORIZONTAL_ALIGNMENT_LEFT, -1, size, color)

func _text_center(text: String, pos: Vector2, size: int, color: Color, width: float) -> void:
	var w = max(1.0, width)
	draw_string(font, Vector2(pos.x - w * 0.5, pos.y), text, HORIZONTAL_ALIGNMENT_CENTER, w, size, color)

func _best_highscore() -> int:
	return max(highscores[MODE_CLASSIC], highscores[MODE_TIME], highscores[MODE_CHAOS])

func _load_save() -> void:
	if not FileAccess.file_exists(SAVE_PATH):
		return
	var file := FileAccess.open(SAVE_PATH, FileAccess.READ)
	if file == null:
		return
	var parsed = JSON.parse_string(file.get_as_text())
	if typeof(parsed) == TYPE_DICTIONARY and parsed.has("highscores"):
		var h = parsed.highscores
		for k in highscores.keys():
			if h.has(k):
				highscores[k] = int(h[k])

func _save() -> void:
	var file := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	if file == null:
		return
	file.store_string(JSON.stringify({"highscores": highscores}))

func _notification(what: int) -> void:
	if what == NOTIFICATION_WM_GO_BACK_REQUEST:
		_handle_back_or_pause()
	elif what == NOTIFICATION_APPLICATION_PAUSED:
		_save()
		if state == "play":
			paused = true
	elif what == NOTIFICATION_APPLICATION_RESUMED:
		_recalculate_layout()

func _exit_tree() -> void:
	_save()
